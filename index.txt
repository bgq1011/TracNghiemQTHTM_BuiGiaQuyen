    let cauHoiHienTai = [];

    function locCauHoi() {
        const chuongDuocChon = document.getElementById('chonChuong').value;
        const loaiDuocChon = document.getElementById('chonLoai').value;

        cauHoiHienTai = danhSachCauHoi.filter(cau => {
            const matchChuong = (chuongDuocChon === 'all') || (cau.chuong == chuongDuocChon);
            const matchLoai = (loaiDuocChon === 'all') || (cau.loai === loaiDuocChon);
            return matchChuong && matchLoai;
        });

        hienThiCauHoi();
    }

    function hienThiCauHoi() {
        const quizContainer = document.getElementById('quiz');
        const stats = document.getElementById('stats');
        
        quizContainer.innerHTML = ''; 
        document.getElementById('result').innerText = ''; 

        if(cauHoiHienTai.length === 0) {
            stats.innerHTML = "Không tìm thấy câu hỏi nào phù hợp với bộ lọc.";
            return;
        }

        stats.innerHTML = `Đang hiển thị <b>${cauHoiHienTai.length}</b> câu hỏi.`;

        cauHoiHienTai.forEach((cau, index) => {
            const div = document.createElement('div');
            div.className = 'question';
            
            const badgeHTML = (cau.loai === 'diem_liet') ? `<span class="badge">[ĐIỂM LIỆT]</span>` : '';
            div.innerHTML = `<h3>${badgeHTML} ${cau.cauHoi}</h3>`;
            
            const ul = document.createElement('ul');
            ul.className = 'options';
            
            cau.luaChon.forEach((lc, vitri) => {
                ul.innerHTML += `
                    <li>
                        <label>
                            <input type="radio" name="cau_${index}" value="${vitri}">
                            ${lc}
                        </label>
                    </li>
                `;
            });
            div.appendChild(ul);
            quizContainer.appendChild(div);
        });
    }

    function kiemTraKetQua() {
        if(cauHoiHienTai.length === 0) return;

        let diem = 0;
        cauHoiHienTai.forEach((cau, index) => {
            const luaChonCuaNguoiDung = document.querySelector(`input[name="cau_${index}"]:checked`);
            if (luaChonCuaNguoiDung && parseInt(luaChonCuaNguoiDung.value) === cau.dapAnDung) {
                diem++;
            }
        });
        
        const tyLe = Math.round((diem / cauHoiHienTai.length) * 100);
